# M1-B 证书与信任安装实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现本地 CA 与按域名签发的叶子证书，并提供 `fastgithub install-cert` 把 CA 装入系统信任库。

**Architecture:** `internal/certs` 负责生成/加载 CA（RSA 2048，`cacert/fastgithub.crt|key`）、按域名签发叶子证书并缓存；`internal/certs/trust.go` 按 `runtime.GOOS` 选择 `certutil`/`security`/`update-ca-certificates`，命令执行与文件复制可注入以便测试；CLI 新增 `install-cert` 子命令。顺带完成 M1-A 交接文档分诊到 M1-B 的 6 项配置边界加固。

**Tech Stack:** Go 1.22（工具链 1.27.1），仅标准库（`crypto/rand`/`crypto/rsa`/`crypto/sha1`/`crypto/tls`/`crypto/x509`/`crypto/x509/pkix`/`encoding/pem`/`math/big`/`net`/`os/exec`/`runtime`/`sync`），无第三方模块；`CGO_ENABLED=0`。

**Spec:** `docs/specs/2026-10-20-fastgithub-go-design.md` §7.3（证书）、§6.4（解析器）、§9.1（单元测试）、§9.3（构建验证）；待办来源 `docs/handoff-2026-10-20.md` §6.4。

## Global Constraints

- 模块路径 `fastgithub-go`，`go.mod` 声明 `go 1.22`，**不出现任何 `require`**。
- 所有提交在仓库根目录进行；提交信息前缀 `feat(certs): `、`fix(config): `、`test(certs): `、`docs: `。
- 面向用户的错误信息用中文；`install-cert` 成功输出中文，失败以 `错误: ` 前缀写到 stderr、退出码 1。
- 每个任务结束时 `go test ./...` 必须全绿，`go vet ./...` 无输出，`gofmt -l .` 无输出。
- 证书文件：私钥 `cacert/fastgithub.key`（PKCS#1 PEM，0600）、证书 `cacert/fastgithub.crt`（PEM，0644）、目录 0700；兼容读取旧版 `cacert/fastgithub.cer`（PEM 或 DER）。
- CA 参数：RSA 2048、`CN=FastGithub`、10 年、`BasicConstraints`（critical，CA=true，pathLen=1）、`KeyUsage`（DigitalSignature|CRLSign|CertSign）、EKU（ServerAuth+ClientAuth）、SKI、SAN=`FastGithub`。
- 叶子参数：RSA 2048、`CN=<domain>`、1 年且不超过 CA、AKI=CA 的 SKI、SAN=域名/机器名/`127.0.0.1`/`::1`、EKU 同 CA。
- 测试不得真正修改系统信任库；信任安装通过注入的命令执行器与文件复制函数验证。

`Decode` 开头加 nil 检查，未知键改为结构体并按行号排序：

```go
type unknownKey struct {
	name string
	line int
}

func Decode(n *Node, out any) error {
	if n == nil {
		return fmt.Errorf("Decode 需要非空语法树")
	}
	rv := reflect.ValueOf(out)
	if rv.Kind() != reflect.Ptr || rv.IsNil() {
		return fmt.Errorf("Decode 需要非空指针")
	}
	var unknown []unknownKey
	if err := decodeInto(n, rv.Elem(), "", &unknown); err != nil {
		return err
	}
	if len(unknown) > 0 {
		sort.Slice(unknown, func(i, j int) bool {
			if unknown[i].line != unknown[j].line {
				return unknown[i].line < unknown[j].line
			}
			return unknown[i].name < unknown[j].name
		})
		parts := make([]string, len(unknown))
		for i, u := range unknown {
			if u.line > 0 {
				parts[i] = fmt.Sprintf("%s (第 %d 行)", u.name, u.line)
			} else {
				parts[i] = u.name
			}
		}
		return fmt.Errorf("未知配置项: %s", strings.Join(parts, ", "))
	}
	return nil
}
```

```go
func parseInteger(s string) (int64, error) {
	body := s
	if body[0] == '+' || body[0] == '-' {
		body = body[1:]
	}
	if body == "" {
		return 0, fmt.Errorf("无法识别的值 %q", s)
	}
	for i := 0; i < len(body); i++ {
		c := body[i]
		if c == '_' {
			if i == 0 || i == len(body)-1 || !isDigit(body[i-1]) || !isDigit(body[i+1]) {
				return 0, fmt.Errorf("整数的下划线必须位于数字之间")
			}
			continue
		}
		if !isDigit(c) {
			return 0, fmt.Errorf("无法识别的值 %q", s)
		}
	}
	return strconv.ParseInt(strings.ReplaceAll(s, "_", ""), 10, 64)
}

func isDigit(c byte) bool { return '0' <= c && c <= '9' }
```

```go
// confFiles 返回 conf.d 下按文件名升序的 .toml 文件；目录不存在视为空。
func confFiles(confDir string) ([]string, error) {
	entries, err := os.ReadDir(confDir)
	if err != nil && !os.IsNotExist(err) {
		return nil, fmt.Errorf("读取 %s 失败: %w", confDir, err)
	}
	names := make([]string, 0, len(entries))
	for _, e := range entries {
		if !e.IsDir() && strings.HasSuffix(e.Name(), ".toml") {
			names = append(names, e.Name())
		}
	}
	sort.Strings(names)
	return names, nil
}
```

```go
package certs

import (
	"crypto/rand"
	"crypto/rsa"
	"crypto/sha1"
	"crypto/tls"
	"crypto/x509"
	"crypto/x509/pkix"
	"encoding/pem"
	"fmt"
	"math/big"
	"os"
	"path/filepath"
	"sync"
	"time"
)

const (
	DefaultDir         = "cacert"
	certFileName       = "fastgithub.crt"
	keyFileName        = "fastgithub.key"
	legacyCertFileName = "fastgithub.cer"
)

// Authority 持有本地 CA 与按域名签发的叶子证书缓存。
type Authority struct {
	dir      string
	certPath string
	keyPath  string
	cert     *x509.Certificate
	key      *rsa.PrivateKey
	now      func() time.Time

	mu    sync.Mutex
	leafs map[string]*tls.Certificate
}

// LoadOrCreate 从 dir 加载 CA；证书与私钥都不存在时生成一套。
func LoadOrCreate(dir string) (*Authority, error) {
	if dir == "" {
		dir = DefaultDir
	}
	certPath := filepath.Join(dir, certFileName)
	keyPath := filepath.Join(dir, keyFileName)
	certExists := fileExists(certPath)
	if !certExists {
		if legacy := filepath.Join(dir, legacyCertFileName); fileExists(legacy) {
			certPath, certExists = legacy, true
		}
	}
	keyExists := fileExists(keyPath)
	switch {
	case certExists && keyExists:
		return load(dir, certPath, keyPath)
	case !certExists && !keyExists:
		return generate(dir, filepath.Join(dir, certFileName), keyPath)
	case certExists:
		return nil, fmt.Errorf("证书 %s 存在但私钥 %s 缺失", certPath, keyPath)
	default:
		return nil, fmt.Errorf("私钥 %s 存在但证书 %s 缺失", keyPath, certPath)
	}
}

func fileExists(path string) bool {
	_, err := os.Stat(path)
	return err == nil
}

func generate(dir, certPath, keyPath string) (*Authority, error) {
	if err := os.MkdirAll(dir, 0700); err != nil {
		return nil, fmt.Errorf("创建证书目录 %s 失败: %w", dir, err)
	}
	_ = os.Chmod(dir, 0700)

	key, err := rsa.GenerateKey(rand.Reader, 2048)
	if err != nil {
		return nil, fmt.Errorf("生成 CA 私钥失败: %w", err)
	}
	now := time.Now()
	tmpl := caTemplate(now, &key.PublicKey)
	der, err := x509.CreateCertificate(rand.Reader, tmpl, tmpl, &key.PublicKey, key)
	if err != nil {
		return nil, fmt.Errorf("签发 CA 证书失败: %w", err)
	}
	cert, err := x509.ParseCertificate(der)
	if err != nil {
		return nil, fmt.Errorf("解析 CA 证书失败: %w", err)
	}
	keyPEM := pem.EncodeToMemory(&pem.Block{Type: "RSA PRIVATE KEY", Bytes: x509.MarshalPKCS1PrivateKey(key)})
	certPEM := pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: der})
	if err := writeFileAtomic(keyPath, keyPEM, 0600); err != nil {
		return nil, err
	}
	if err := writeFileAtomic(certPath, certPEM, 0644); err != nil {
		return nil, err
	}
	return newAuthority(dir, certPath, keyPath, cert, key), nil
}

func load(dir, certPath, keyPath string) (*Authority, error) {
	keyData, err := os.ReadFile(keyPath)
	if err != nil {
		return nil, fmt.Errorf("读取私钥 %s 失败: %w", keyPath, err)
	}
	key, err := parsePrivateKey(keyData)
	if err != nil {
		return nil, fmt.Errorf("解析私钥 %s 失败: %w", keyPath, err)
	}
	certData, err := os.ReadFile(certPath)
	if err != nil {
		return nil, fmt.Errorf("读取证书 %s 失败: %w", certPath, err)
	}
	cert, err := parseCertificate(certData)
	if err != nil {
		return nil, fmt.Errorf("解析证书 %s 失败: %w", certPath, err)
	}
	if !cert.IsCA {
		return nil, fmt.Errorf("证书 %s 不是 CA", certPath)
	}
	if !publicKeysEqual(cert.PublicKey, &key.PublicKey) {
		return nil, fmt.Errorf("证书 %s 与私钥 %s 不匹配", certPath, keyPath)
	}
	return newAuthority(dir, certPath, keyPath, cert, key), nil
}

func newAuthority(dir, certPath, keyPath string, cert *x509.Certificate, key *rsa.PrivateKey) *Authority {
	return &Authority{
		dir:      dir,
		certPath: certPath,
		keyPath:  keyPath,
		cert:     cert,
		key:      key,
		now:      time.Now,
		leafs:    map[string]*tls.Certificate{},
	}
}

func caTemplate(now time.Time, pub *rsa.PublicKey) *x509.Certificate {
	serial, _ := rand.Int(rand.Reader, new(big.Int).Lsh(big.NewInt(1), 128))
	return &x509.Certificate{
		SerialNumber:          serial,
		Subject:               pkix.Name{CommonName: "FastGithub"},
		NotBefore:             now.Add(-time.Minute),
		NotAfter:              now.AddDate(10, 0, 0),
		KeyUsage:              x509.KeyUsageDigitalSignature | x509.KeyUsageCertSign | x509.KeyUsageCRLSign,
		ExtKeyUsage:           []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth, x509.ExtKeyUsageClientAuth},
		BasicConstraintsValid: true,
		IsCA:                  true,
		MaxPathLen:            1,
		SubjectKeyId:          subjectKeyID(pub),
		DNSNames:              []string{"FastGithub"},
	}
}

func subjectKeyID(pub *rsa.PublicKey) []byte {
	der, err := x509.MarshalPKIXPublicKey(pub)
	if err != nil {
		return nil
	}
	sum := sha1.Sum(der)
	return sum[:]
}

// Certificate 返回 CA 证书。
func (a *Authority) Certificate() *x509.Certificate { return a.cert }

// CertPEM 返回 CA 证书的 PEM 编码。
func (a *Authority) CertPEM() []byte {
	return pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: a.cert.Raw})
}

// CertPath 返回 CA 证书文件路径。
func (a *Authority) CertPath() string { return a.certPath }

// KeyPath 返回 CA 私钥文件路径。
func (a *Authority) KeyPath() string { return a.keyPath }
```

```go
// writeFileAtomic 先写同目录临时文件再重命名，避免留下半截证书。
func writeFileAtomic(path string, data []byte, perm os.FileMode) error {
	dir := filepath.Dir(path)
	tmp, err := os.CreateTemp(dir, "."+filepath.Base(path)+".*")
	if err != nil {
		return fmt.Errorf("创建临时文件失败: %w", err)
	}
	tmpName := tmp.Name()
	defer os.Remove(tmpName)
	if _, err := tmp.Write(data); err != nil {
		tmp.Close()
		return fmt.Errorf("写入 %s 失败: %w", path, err)
	}
	if err := tmp.Close(); err != nil {
		return fmt.Errorf("关闭 %s 失败: %w", path, err)
	}
	if err := os.Chmod(tmpName, perm); err != nil {
		return fmt.Errorf("设置 %s 权限失败: %w", path, err)
	}
	if err := os.Rename(tmpName, path); err != nil {
		return fmt.Errorf("保存 %s 失败: %w", path, err)
	}
	return nil
}

func parsePrivateKey(data []byte) (*rsa.PrivateKey, error) {
	block, _ := pem.Decode(data)
	if block == nil {
		return nil, fmt.Errorf("不是 PEM 格式")
	}
	if key, err := x509.ParsePKCS1PrivateKey(block.Bytes); err == nil {
		return key, nil
	}
	parsed, err := x509.ParsePKCS8PrivateKey(block.Bytes)
	if err != nil {
		return nil, err
	}
	key, ok := parsed.(*rsa.PrivateKey)
	if !ok {
		return nil, fmt.Errorf("不是 RSA 私钥")
	}
	return key, nil
}

func parseCertificate(data []byte) (*x509.Certificate, error) {
	if block, _ := pem.Decode(data); block != nil && block.Type == "CERTIFICATE" {
		data = block.Bytes
	}
	return x509.ParseCertificate(data)
}

func publicKeysEqual(a, b any) bool {
	ka, ok := a.(*rsa.PublicKey)
	if !ok {
		return false
	}
	kb, ok := b.(*rsa.PublicKey)
	if !ok {
		return false
	}
	return ka.E == kb.E && ka.N.Cmp(kb.N) == 0
}
```

```go
package certs

import (
	"crypto/rand"
	"crypto/rsa"
	"crypto/tls"
	"crypto/x509"
	"crypto/x509/pkix"
	"fmt"
	"math/big"
	"net"
	"os"
	"time"
)

const (
	maxLeafCache    = 1024
	leafRenewBefore = 24 * time.Hour
	leafLifetime    = 365 * 24 * time.Hour
)

// LeafFor 返回 domain 对应的叶子证书，命中缓存时直接复用。
func (a *Authority) LeafFor(domain string) (*tls.Certificate, error) {
	if domain == "" {
		return nil, fmt.Errorf("域名不能为空")
	}
	a.mu.Lock()
	defer a.mu.Unlock()
	now := a.now()
	if c, ok := a.leafs[domain]; ok && c.Leaf != nil && now.Add(leafRenewBefore).Before(c.Leaf.NotAfter) {
		return c, nil
	}
	c, err := a.newLeaf(domain, now)
	if err != nil {
		return nil, err
	}
	if len(a.leafs) >= maxLeafCache {
		a.sweepLocked(now)
	}
	a.leafs[domain] = c
	return c, nil
}

func (a *Authority) newLeaf(domain string, now time.Time) (*tls.Certificate, error) {
	key, err := rsa.GenerateKey(rand.Reader, 2048)
	if err != nil {
		return nil, fmt.Errorf("生成 %s 的私钥失败: %w", domain, err)
	}
	serial, err := rand.Int(rand.Reader, new(big.Int).Lsh(big.NewInt(1), 128))
	if err != nil {
		return nil, fmt.Errorf("生成序列号失败: %w", err)
	}
	notAfter := now.Add(leafLifetime)
	if notAfter.After(a.cert.NotAfter) {
		notAfter = a.cert.NotAfter
	}
	tmpl := &x509.Certificate{
		SerialNumber:          serial,
		Subject:               pkix.Name{CommonName: domain},
		NotBefore:             now.Add(-time.Minute),
		NotAfter:              notAfter,
		KeyUsage:              x509.KeyUsageDigitalSignature | x509.KeyUsageKeyEncipherment,
		ExtKeyUsage:           []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth, x509.ExtKeyUsageClientAuth},
		BasicConstraintsValid: true,
		AuthorityKeyId:        a.cert.SubjectKeyId,
	}
	addSAN(tmpl, domain)
	der, err := x509.CreateCertificate(rand.Reader, tmpl, a.cert, &key.PublicKey, a.key)
	if err != nil {
		return nil, fmt.Errorf("签发 %s 的证书失败: %w", domain, err)
	}
	leaf, err := x509.ParseCertificate(der)
	if err != nil {
		return nil, fmt.Errorf("解析 %s 的证书失败: %w", domain, err)
	}
	return &tls.Certificate{
		Certificate: [][]byte{der, a.cert.Raw},
		PrivateKey:  key,
		Leaf:        leaf,
	}, nil
}

func addSAN(t *x509.Certificate, domain string) {
	if ip := net.ParseIP(domain); ip != nil {
		t.IPAddresses = append(t.IPAddresses, ip)
	} else {
		t.DNSNames = append(t.DNSNames, domain)
	}
	if host, err := os.Hostname(); err == nil && host != "" && host != domain {
		t.DNSNames = append(t.DNSNames, host)
	}
	for _, raw := range []string{"127.0.0.1", "::1"} {
		ip := net.ParseIP(raw)
		if !containsIP(t.IPAddresses, ip) {
			t.IPAddresses = append(t.IPAddresses, ip)
		}
	}
}

func containsIP(ips []net.IP, want net.IP) bool {
	for _, ip := range ips {
		if ip.Equal(want) {
			return true
		}
	}
	return false
}

// sweepLocked 清理已过期的缓存；调用方必须持有 a.mu。
func (a *Authority) sweepLocked(now time.Time) {
	for domain, c := range a.leafs {
		if c.Leaf == nil || !now.Before(c.Leaf.NotAfter) {
			delete(a.leafs, domain)
		}
	}
}
```

```go
package certs

import (
	"fmt"
	"io"
	"os"
	"os/exec"
	"path/filepath"
	"runtime"
	"strings"
)

type commandRunner func(name string, args ...string) error
type fileCopier func(src, dst string) error

// Install 把 certPath 指向的 CA 证书加入系统信任库，需要管理员/root 权限。
func Install(certPath string) error {
	return install(certPath, runtime.GOOS, runCommand, copyFile)
}

func install(certPath, goos string, run commandRunner, copyFile fileCopier) error {
	switch goos {
	case "windows":
		if err := run("certutil", "-addstore", "-f", "Root", certPath); err != nil {
			return fmt.Errorf("certutil 导入失败（可能需要管理员权限）: %w", err)
		}
	case "darwin":
		if err := run("security", "add-trusted-cert", "-d", "-r", "trustRoot",
			"-k", "/Library/Keychains/System.keychain", certPath); err != nil {
			return fmt.Errorf("security 导入失败（可能需要 sudo）: %w", err)
		}
	case "linux":
		const dest = "/usr/local/share/ca-certificates/fastgithub.crt"
		if err := copyFile(certPath, dest); err != nil {
			return fmt.Errorf("复制证书到 %s 失败（需要 root）: %w", dest, err)
		}
		if err := run("update-ca-certificates"); err != nil {
			return fmt.Errorf("update-ca-certificates 失败（可能需要 root）: %w", err)
		}
	default:
		return fmt.Errorf("不支持的平台 %s，请手动导入证书 %s", goos, certPath)
	}
	return nil
}

func runCommand(name string, args ...string) error {
	out, err := exec.Command(name, args...).CombinedOutput()
	if err != nil {
		if msg := strings.TrimSpace(string(out)); msg != "" {
			return fmt.Errorf("%w: %s", err, msg)
		}
		return err
	}
	return nil
}

func copyFile(src, dst string) error {
	in, err := os.Open(src)
	if err != nil {
		return err
	}
	defer in.Close()
	if err := os.MkdirAll(filepath.Dir(dst), 0755); err != nil {
		return err
	}
	out, err := os.OpenFile(dst, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, 0644)
	if err != nil {
		return err
	}
	if _, err := io.Copy(out, in); err != nil {
		out.Close()
		return err
	}
	return out.Close()
}
```

`main.go`：

```go
import (
	"flag"
	"fmt"
	"io"
	"os"

	"fastgithub-go/internal/app"
	"fastgithub-go/internal/certs"
	"fastgithub-go/internal/config"
)

const usage = `用法: fastgithub <命令>

命令:
  config validate [--path 文件]        校验配置
  install-cert [--dir 目录]            生成 CA 并加入系统信任库（需管理员/root）
  version                              打印版本信息
`

// installCert 可在测试中替换，避免真正修改系统信任库。
var installCert = certs.Install

func run(args []string, stdout, stderr io.Writer) int {
	if len(args) == 0 {
		return usageError(stderr)
	}
	switch args[0] {
	case "version":
		if len(args) != 1 {
			return usageError(stderr)
		}
		fmt.Fprintln(stdout, app.BuildInfo())
		return 0
	case "config":
		if err := runConfig(args[1:], stdout); err != nil {
			fmt.Fprintln(stderr, "错误: "+err.Error())
			return 1
		}
		return 0
	case "install-cert":
		if err := runInstallCert(args[1:], stdout); err != nil {
			fmt.Fprintln(stderr, "错误: "+err.Error())
			return 1
		}
		return 0
	default:
		return usageError(stderr)
	}
}

func runInstallCert(args []string, stdout io.Writer) error {
	fs := flag.NewFlagSet("install-cert", flag.ContinueOnError)
	fs.SetOutput(io.Discard)
	dir := fs.String("dir", certs.DefaultDir, "CA 证书目录")
	if err := fs.Parse(args); err != nil {
		return fmt.Errorf("参数错误: %w", err)
	}
	if fs.NArg() != 0 {
		return fmt.Errorf("用法: fastgithub install-cert [--dir 目录]")
	}
	a, err := certs.LoadOrCreate(*dir)
	if err != nil {
		return err
	}
	fmt.Fprintf(stdout, "CA 证书: %s\n", a.CertPath())
	if err := installCert(a.CertPath()); err != nil {
		return err
	}
	fmt.Fprintln(stdout, "已加入系统信任库")
	return nil
}
```
